void unordered_set_diff() {
    unordered_set<int> us;

    /* 构造 */
    unordered_set<int> us1;
    unordered_set<int> us2 = {1, 2, 3};

    /* 哈希相关（⚠ 仅无序容器有） */
    us.bucket_count();              // 桶数量
    us.bucket(3);                   // 元素 3 在哪个桶
    us.load_factor();               // 当前负载因子
    us.max_load_factor();           // 最大负载因子（默认 1.0）
    us.max_load_factor(0.7);        // 降低冲突（竞赛很少调）

    /* 预分配（非常重要！） */
    us.reserve(100000);             // 直接避免频繁 rehash
    us.rehash(200000);              // 强制桶数量

    /* ⚠ 不支持的操作 */

    // 自动排序 ❌

}

/*=====================================
=        unordered_multiset           =
=====================================*/
void unordered_multiset_diff() {
    unordered_multiset<int> ums;

    /* 允许重复 */
    ums.insert(1);
    ums.insert(1);

    /* 删除 */
    ums.erase(1);                   // 删除所有 ==1
    auto it = ums.find(1);
    if (it != ums.end()) ums.erase(it); // 删除一个

    /* ⚠ 顺序无意义 */
    // *ums.begin() ≠ 最小值
}

void unordered_map_diff() {
    unordered_map<int, int> ump;

    /* 构造 */
    unordered_map<int, int> ump1;

    /* 预分配（竞赛关键优化） */
    ump.reserve(200000);
    ump.max_load_factor(0.7);

    /* 插入（同 map，不重复写） */
    ump.emplace(1, 10);
    ump[2] = 20;

    /* 访问 */
    ump[100]++;                     // 不存在 → 创建

    /* ⚠ 与 map 不同 */
    // 1. 遍历顺序随机
    // 2. 不支持按 key 区间查询
    // 3. 迭代器失效更频繁（rehash）

    /* 典型用途 */
    // - 高频计数
    // - 状态哈希
    // - 映射但不关心顺序
}

/* 防止 hack 的常用模板 */
struct custom_hash {
    static uint64_t splitmix64(uint64_t x) {
        x += 0x9e3779b97f4a7c15;
        x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9;
        x = (x ^ (x >> 27)) * 0x94d049bb133111eb;
        return x ^ (x >> 31);
    }

    size_t operator()(uint64_t x) const {
        static const uint64_t FIXED_RANDOM =
            chrono::steady_clock::now().time_since_epoch().count();
        return splitmix64(x + FIXED_RANDOM);
    }
};

/* 使用示例 */
void custom_hash_usage() {
    unordered_map<long long, int, custom_hash> ump;
    unordered_set<long long, custom_hash> us;
}
