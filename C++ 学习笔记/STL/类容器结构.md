/*=====================================

void bitset_all() {
    /* 构造（编译期定长！） */
    bitset<10> b1;                  // 全 0
    bitset<10> b2(string("10101")); // 从字符串构造
    bitset<10> b3(5);               // 从整数构造（二进制）

    /* 访问 */
    b1[0];                          // 第 0 位（低位）
    b1.test(0);                     // 是否为 1

    /* 修改 */
    b1.set();                       // 全部置 1
    b1.set(3);                      // 第 3 位设为 1
    b1.reset();                     // 全部清 0
    b1.reset(3);                    // 第 3 位清 0
    b1.flip();                      // 全部翻转
    b1.flip(2);                     // 第 2 位翻转

    /* 查询 */
    b1.any();                       // 是否存在 1
    b1.none();                      // 是否全 0
    b1.all();                       // 是否全 1
    b1.count();                     // 1 的个数
    b1.size();                      // 位数

    /* 位运算（逐位） */
    bitset<10> a("101010");
    bitset<10> c("001111");
    a & c;
    a | c;
    a ^ c;
    ~a;
    a <<= 2;
    a >>= 1;

    /* 转换 */
    a.to_string();                  // 转 string
    a.to_ulong();                   // 转 unsigned long
    a.to_ullong();                  // 转 unsigned long long

    /* 特点 */
    // 1. 编译期定长（不能动态）
    // 2. 常数极小，比 vector<bool> 快
    // 3. 非常适合状态压缩 DP
}

void pair_all() {
    /* 构造 */
    pair<int, int> p1;
    pair<int, int> p2 = {1, 2};
    pair<int, int> p3(3, 4);
    auto p4 = make_pair(5, 6);

    /* 访问 */
    p2.first;
    p2.second;

    /* 比较（字典序） */
    // 先比 first，再比 second
    pair<int,int> a = {1, 2};
    pair<int,int> b = {1, 3};
    bool ok = a < b;                // true

    /* 结构化绑定（C++17） */
    auto [x, y] = p3;

    /* 常见用途 */
    // - 坐标
    // - (权值, 点)
    // - map / set 的元素

    /* 在 STL 中的地位 */
    // priority_queue<pair<...>>
    // map<Key, Value> 的元素本质是 pair<const Key, Value>
}

void tuple_all() {
    /* 构造 */
    tuple<int, int, int> t1;
    tuple<int, int, string> t2(1, 2, "abc");
    auto t3 = make_tuple(3, 4, 5);

    /* 访问（编译期索引） */
    get<0>(t2);
    get<1>(t2);
    get<2>(t2);

    /* 修改 */
    get<0>(t2) = 10;

    /* 结构化绑定（推荐） */
    auto [a, b, c] = t2;

    /* 比较（字典序） */
    tuple<int,int,int> x = {1,2,3};
    tuple<int,int,int> y = {1,2,4};
    bool ok = x < y;                // true

    /* tie 用法 */
    int u, v;
    string s;
    tie(u, v, s) = t2;

    /* 忽略部分值 */
    tie(u, ignore, s) = t2;

    /* 常见用途 */
    // - 多关键字排序
    // - 返回多个值
    // - 复杂状态存储（但竞赛慎用）
}

void bitset_dp_example() {
    const int N = 20;
    bitset<N> mask;
    mask.set(3);
    if (mask.test(3)) {
        // 状态存在
    }
}

    pq.push({0, 1}); // {距离, 点}
