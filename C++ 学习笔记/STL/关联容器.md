void set_all() {
    set<int> s;

    /* 构造 */
    set<int> s1;
    set<int> s2 = {1, 3, 5};
    set<int> s3(s2.begin(), s2.end());

    /* 插入（自动去重） */
    s.insert(3);
    s.insert({1, 2, 4});

    /* 删除 */
    s.erase(3);                     // 删除值
    s.erase(s.begin());             // 删除迭代器
    s.clear();

    /* 查找 */
    s.find(3);                      // 找不到返回 end()
    s.count(3);                     // 只可能是 0 / 1
    s.contains(3);                  // C++20（竞赛很少用）

    /* 范围查找（非常重要） */
    s.lower_bound(3);               // >= 3 的第一个
    s.upper_bound(3);               // >  3 的第一个
    s.equal_range(3);               // {lower, upper}

    /* 访问 */
    *s.begin();                     // 最小值
    *prev(s.end());                 // 最大值

    /* 迭代器 */
    s.begin();
    s.end();
    s.rbegin();
    s.rend();

    /* 容量 */
    s.size();
    s.empty();

    /* 自定义排序 */
    set<int, greater<int>> s_desc;  // 降序

    /* 典型用途 */
    // - 去重 + 保序
    // - 动态维护最小 / 最大值
    // - 区间存在性判断
}

void map_all() {
    map<int, int> mp;

    /* 构造 */
    map<int, int> mp1;
    map<int, int> mp2 = {{1, 2}, {3, 4}};

    /* 插入 */
    mp.insert({1, 10});
    mp.emplace(2, 20);              // 更快
    mp[3] = 30;                     // ⚠ 若不存在会自动创建

    /* 访问 */
    mp[1];                          // 若 key 不存在，插入 value=0
    mp.at(1);                       // 不存在会抛异常（竞赛少用）

    /* 查找 */
    mp.find(1);
    mp.count(1);                    // 0 / 1
    mp.contains(1);                 // C++20

    /* 删除 */
    mp.erase(1);
    mp.erase(mp.begin());
    mp.clear();

    /* 范围查找 */
    mp.lower_bound(2);              // key >= 2
    mp.upper_bound(2);              // key >  2
    mp.equal_range(2);

    /* 迭代 */
    for (auto &[k, v] : mp) {
        // k 是 key，v 是 value
    }

    /* 容量 */
    mp.size();
    mp.empty();

    /* 自定义 key 排序 */
    map<int, int, greater<int>> mp_desc;

    /* 典型用途 */
    // - 计数
    // - 离散化映射
    // - 按 key 有序遍历
}
