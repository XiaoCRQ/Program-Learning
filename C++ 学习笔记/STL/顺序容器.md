void vector_all() {
    vector<int> v;

    /* 构造 */
    vector<int> v1;                 // 空
    vector<int> v2(5);              // 5 个 0
    vector<int> v3(5, 3);           // 5 个 3
    vector<int> v4(v3);             // 拷贝
    vector<int> v5(v3.begin(), v3.end());

    /* 访问 */
    v[0];                           // 不检查越界
    v.at(0);                        // 检查越界
    v.front();                      // 第一个
    v.back();                       // 最后一个
    v.data();                       // 底层数组指针（竞赛少用）

    /* 迭代器 */
    v.begin();
    v.end();
    v.rbegin();
    v.rend();
    v.cbegin();
    v.cend();

    /* 容量 */
    v.size();
    v.capacity();
    v.empty();
    v.reserve(100);                 // 预分配（重要！防止扩容）
    v.shrink_to_fit();              // 释放多余容量（不保证）

    /* 修改 */
    v.push_back(1);
    v.pop_back();
    v.insert(v.begin() + 1, 10);
    v.insert(v.begin(), 3, 7);      // 插入 3 个 7
    v.insert(v.begin(), v3.begin(), v3.end());
    v.erase(v.begin());
    v.erase(v.begin(), v.begin() + 3);
    v.clear();
    v.resize(10);                   // 多的补 0
    v.resize(10, -1);               // 多的补 -1
    v.assign(5, 100);               // 重置为 5 个 100
    v.swap(v3);                     // O(1) 交换

    /* 算法配合 */
    sort(v.begin(), v.end());
    reverse(v.begin(), v.end());
    lower_bound(v.begin(), v.end(), 5);
}

void deque_all() {
    deque<int> d;

    /* 构造 */
    deque<int> d1;
    deque<int> d2(5);
    deque<int> d3(5, 2);

    /* 访问 */
    d[0];
    d.at(0);
    d.front();
    d.back();

    /* 迭代器 */
    d.begin();
    d.end();

    /* 容量 */
    d.size();
    d.empty();

    /* 修改（双端） */
    d.push_back(1);
    d.push_front(2);
    d.pop_back();
    d.pop_front();

    /* 插入删除 */
    d.insert(d.begin(), 10);
    d.erase(d.begin());
    d.clear();

    /* 特点 */
    // 1. 支持随机访问
    // 2. 头尾插入 O(1)
    // 3. 中间插入慢
}

void list_all() {
    list<int> l;

    /* 构造 */
    list<int> l1;
    list<int> l2(5, 3);

    /* 访问（无随机访问！） */
    l.front();
    l.back();

    /* 迭代器 */
    auto it = l.begin();
    advance(it, 2);                 // 手动前进

    /* 容量 */
    l.size();
    l.empty();

    /* 修改（O(1)） */
    l.push_back(1);
    l.push_front(2);
    l.pop_back();
    l.pop_front();
    l.insert(l.begin(), 10);
    l.erase(l.begin());
    l.clear();

    /* list 特有 */
    l.remove(3);                    // 删除值 == 3 的元素
    l.remove_if([](int x){ return x % 2 == 0; });
    l.unique();                     // 去重（相邻）
    l.sort();                       // 自带排序
    l.reverse();                    // 反转
    l.splice(l.begin(), l2);        // 拼接 l2 到 l（O(1)）

    /* 特点 */
    // 1. 双向链表
    // 2. 不支持下标
    // 3. 插删极快
}

void forward_list_all() {
    forward_list<int> fl;

    /* 构造 */
    forward_list<int> fl1;
    forward_list<int> fl2(5, 1);

    /* 访问 */
    fl.front();

    /* 迭代器（只有 forward） */
    fl.begin();
    fl.end();
    fl.before_begin();              // 重要！

    /* 修改 */
    fl.push_front(1);
    fl.pop_front();
    fl.insert_after(fl.before_begin(), 10);
    fl.erase_after(fl.before_begin());
    fl.clear();

    /* 特有 */
    fl.remove(3);
    fl.remove_if([](int x){ return x < 0; });
    fl.unique();
    fl.sort();
    fl.reverse();

    /* 特点 */
    // 1. 单向链表
    // 2. 内存最省
    // 3. 竞赛几乎不用
}

    a.begin();

    /* 容量 */
    a.size();
    a.empty();                      // size=0 才 true（几乎不用）

    /* 特点 */
    // 1. 编译期定长
    // 2. 栈分配
    // 3. 比 vector 更安全更快
}

void string_all() {
    string s;

    /* 构造 */
    string s1 = "abc";
    string s2(5, 'x');
    string s3(s1);
    string s4(s1.begin(), s1.end());

    /* 访问 */
    s[0];
    s.at(0);
    s.front();
    s.back();
    s.c_str();                      // const char*
    s.data();                       // char*

    /* 容量 */
    s.size();
    s.length();
    s.capacity();
    s.empty();
    s.reserve(100);
    s.shrink_to_fit();

    /* 修改 */
    s += "abc";
    s.push_back('x');
    s.pop_back();
    s.append("xyz");
    s.insert(1, "HELLO");
    s.erase(1, 3);
    s.clear();
    s.replace(0, 2, "AA");

    /* 查找 */
    s.find("abc");
    s.rfind("abc");
    s.find_first_of("abc");
    s.find_last_of("abc");
    s.find_first_not_of("abc");

    /* 子串 */
    s.substr(1, 3);

    /* 比较 */
    s.compare("abc");

    /* 转换 */
    stoi("123");
    stol("123");
    stoll("123");
    to_string(123);

    /* 特点 */
    // 1. 本质是 vector<char>
    // 2. 竞赛字符串核心容器
}
