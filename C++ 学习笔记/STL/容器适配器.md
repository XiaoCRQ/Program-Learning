void stack_all() {
    stack<int> st;

    /* 构造 */
    stack<int> st1;
    stack<int, vector<int>> st2;    // 指定底层容器
    stack<int, deque<int>> st3;     // 默认是 deque

    st.push(1);                     // 入栈
    st.pop();                       // 出栈（不返回）
    st.top();                       // 查看栈顶
    st.size();
    st.empty();

    /* ⚠ 注意 */
    // 1. pop() 不返回元素

    /* 典型用途 */
    // - 括号匹配

    // - 单调栈
}

/* 单调栈模板 */
void mono_stack_example(vector<int>& a) {
    stack<int> st;
    for (int i = 0; i < (int)a.size(); i++) {
        while (!st.empty() && a[st.top()] > a[i]) {
            st.pop();
        }
        st.push(i);
    }
}

/* BFS 模板 */
void bfs_example(int s, vector<vector<int>>& g) {
    queue<int> q;
    vector<int> vis(g.size(), 0);
    q.push(s);
    vis[s] = 1;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : g[u]) {
            if (!vis[v]) {
                vis[v] = 1;
                q.push(v);
            }
        }
    }
}

/* Dijkstra 模板 */
void dijkstra_example(int s, vector<vector<pair<int,int>>>& g) {
    const long long INF = 1e18;
    vector<long long> dist(g.size(), INF);
    priority_queue<pair<long long,int>,
        vector<pair<long long,int>>,
        greater<pair<long long,int>>> pq;

    dist[s] = 0;
    pq.push({0, s});
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d != dist[u]) continue;
        for (auto [v, w] : g[u]) {
            if (dist[v] > d + w) {
                dist[v] = d + w;
                pq.push({dist[v], v});
            }
        }
    }
}
