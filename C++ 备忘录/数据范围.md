// C++ 常见整数类型数据范围备忘录（含编译器扩展类型）

// 标准有符号整型
signed char      c;        // -2^7 ~ 2^7-1               = -128 ~ 127
short            s;        // -2^15 ~ 2^15-1             = -32,768 ~ 32,767
int              i;        // -2^31 ~ 2^31-1             ≈ -2.15e9 ~ 2.15e9
long             l;        // -2^31 ~ 2^31-1 (LP32/ILP32) 或 -2^63 ~ 2^63-1 (LP64) → 通常按 -2^31 ~ 2^31-1 = -2.15e9 ~ 2.15e9
long long        ll;       // -2^63 ~ 2^63-1             ≈ -9.22e18 ~ 9.22e18

// 标准无符号整型
unsigned char    uc;       // 0 ~ 2^8-1                  = 0 ~ 255
unsigned short   us;       // 0 ~ 2^16-1                 = 0 ~ 65,535
unsigned int     ui;       // 0 ~ 2^32-1                 ≈ 4.29e9
unsigned long    ul;       // 0 ~ 2^32-1 或 2^64-1       → 通常按 0 ~ 2^32-1 ≈ 4.29e9
unsigned long long ull;    // 0 ~ 2^64-1                 ≈ 1.84e19

// 编译器扩展：GCC/Clang 特有（非标准，但广泛支持）
__int128         i128;     // -2^127 ~ 2^127-1           ≈ -1.70e38 ~ 1.70e38 （2^127 ≈ 1.701e38）
unsigned __int128 u128;    // 0 ~ 2^128-1                ≈ 3.40e38 （2^128 ≈ 3.403e38）

// 浮点类型（IEEE 754）
float            f;        // ±1.18e−38 ~ ±3.40e38       （约 7 位有效数字）
double           d;        // ±2.23e−308 ~ ±1.80e308     （约 15–17 位有效数字）
long double      ld;       // 平台相关（x86 通常为 80-bit：±3.37e−4932 ~ ±1.18e4932，约 18–19 位有效数字）

// 注：
// - 所有 2^n 转换为 10 的幂使用近似：2^10 ≈ 1.024e3 ⇒ log10(2) ≈ 0.3010
//   因此 2^n ≈ 10^(n × 0.3010)
// - 例如：2^63 ≈ 10^(63×0.3010) ≈ 10^18.963 ≈ 9.22e18
// - __int128 不支持标准 I/O（需自行实现输入输出）
